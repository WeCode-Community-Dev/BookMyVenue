function SectionCard({
  icon: Icon,
  title,
  subtitle,
  children,
  className,
}: {
  icon: React.ElementType;
  title: string;
  subtitle?: string;
  children: React.ReactNode;
  className?: string;
}) {
  return (
    <div
      className={cn(
        "bg-background rounded-2xl border border-border p-6 space-y-5",
        className,
      )}
    >
      <div className="flex items-center gap-3">
        <div className="flex items-center justify-center w-9 h-9 rounded-xl bg-primary/10 shrink-0">
          <Icon className="w-4.5 h-4.5 text-primary" strokeWidth={2} />
        </div>

function FieldGroup({ children }: { children: React.ReactNode }) {
  return <div className="space-y-4">{children}</div>;
}

function Field({
  label,
  required,
  children,
}: {
  label: string;
  required?: boolean;
  children: React.ReactNode;
}) {
  return (
    <div className="space-y-1.5">
      <Label className="text-xs font-semibold text-foreground/80">
        {label}
        {required && <span className="text-destructive ml-0.5">*</span>}
      </Label>
      {children}
    </div>
  );
}

function AmenityChip({
  id,
  label,
  icon: Icon,
  selected,
  onToggle,
}: {
  id: string;
  label: string;
  icon: React.ElementType;
  selected: boolean;
  onToggle: (id: string) => void;
}) {
  return (
    <button
      type="button"
      onClick={() => onToggle(id)}
      className={cn(
        "flex items-center gap-2 px-3.5 py-2 rounded-full border text-sm font-medium transition-all",
        selected

          : "border-border bg-background text-muted-foreground hover:border-primary/50 hover:text-foreground",
      )}
    >
      <Icon className="w-3.5 h-3.5" />
      {label}
      {selected && <Check className="w-3 h-3 ml-0.5" />}
    </button>
  );
}

// ─── Step 2 ───────────────────────────────────────────────────────────────────

function MediaFeaturesStep({
  form,
  setForm,
}: {
  form: VenueFormData;
  setForm: React.Dispatch<React.SetStateAction<VenueFormData>>;
}) {
  const set = <K extends keyof VenueFormData>(k: K, v: VenueFormData[K]) =>
    setForm((f) => ({ ...f, [k]: v }));

  const handleFiles = (e: React.ChangeEvent<HTMLInputElement>) => {
    const files = Array.from(e.target.files ?? []).slice(0, 4 - form.imageFiles.length);
    const previews = files.map((f) => URL.createObjectURL(f));
    setForm((prev) => ({
      ...prev,
      imageFiles: [...prev.imageFiles, ...files].slice(0, 4),
      imagePreviews: [...prev.imagePreviews, ...previews].slice(0, 4),
    }));
  };

  const removeImage = (i: number) => {
    URL.revokeObjectURL(form.imagePreviews[i]);
    setForm((prev) => ({
      ...prev,
      imageFiles: prev.imageFiles.filter((_, idx) => idx !== i),
      imagePreviews: prev.imagePreviews.filter((_, idx) => idx !== i),
    }));
  };

  const toggleAmenity = (id: string) => {
    setForm((prev) => ({
      ...prev,
      amenities: prev.amenities.includes(id)
        ? prev.amenities.filter((a) => a !== id)
        : [...prev.amenities, id],
    }));
  };

  return (
    <div className="space-y-5">
      <SectionCard
        icon={ImageIcon}
        title="Space photos"
        subtitle="Upload up to 4 photos — first photo is the cover"
      >
        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
          {form.imagePreviews.map((src, i) => (
            <div
              key={i}
              className="relative aspect-[4/3] rounded-xl overflow-hidden border border-border group"
            >
              <img
                src={src}
                alt={`Upload ${i + 1}`}
                className="w-full h-full object-cover"
              />
              {i === 0 && (
                <div className="absolute top-2 left-2">
                  <Badge className="text-[10px] px-1.5 py-0.5 bg-primary text-primary-foreground">
                    Cover
                  </Badge>
                </div>
              )}
              <button
                type="button"
                onClick={() => removeImage(i)}
                className="absolute top-2 right-2 w-6 h-6 rounded-full bg-black/60 text-white flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity"
              >
                <Trash2 className="w-3 h-3" />
              </button>
            </div>
          ))}
          {form.imagePreviews.length < 4 && (
            <label className="aspect-[4/3] rounded-xl border-2 border-dashed border-border flex flex-col items-center justify-center gap-2 cursor-pointer hover:border-primary/50 hover:bg-primary/5 transition-colors">
              <Upload className="w-5 h-5 text-muted-foreground" />
              <span className="text-xs text-muted-foreground">Add photo</span>
              <input
                type="file"
                accept="image/*"
                multiple
                className="hidden"
                onChange={handleFiles}
              />
            </label>
          )}
        </div>
      </SectionCard>

      <SectionCard
        icon={Sparkles}
        title="Amenities"
        subtitle="Select everything available at your space"
      >
        <div className="flex flex-wrap gap-2">
          {AMENITIES.map((a) => (
            <AmenityChip
              key={a.id}
              {...a}
              selected={form.amenities.includes(a.id)}
              onToggle={toggleAmenity}
            />
          ))}
        </div>
      </SectionCard>

      <SectionCard
        icon={DollarSign}
        title="Capacity & extras"
        subtitle="Help guests plan their visit"
      >
        <FieldGroup>
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <Field label="Seating capacity (guests)" required>
              <Input
                type="number"
                min={1}
                placeholder="e.g. 80"
                value={form.seatingCapacity}
                onChange={(e) =>
                  set(
                    "seatingCapacity",
                    e.target.value === "" ? "" : Number(e.target.value),
                  )
                }
                className="rounded-xl"
              />
            </Field>
          </div>
          <div className="flex items-center justify-between py-3 border border-border rounded-xl px-4">
            <div>
              <p className="text-sm font-medium text-foreground">Parking available</p>
              <p className="text-xs text-muted-foreground">On-site or adjacent parking for guests</p>
            </div>
            <Switch
              checked={form.parking}
              onCheckedChange={(v) => set("parking", v)}
            />
          </div>
          <Field label="Additional features or notes">
            <Textarea
              rows={2}
              placeholder="e.g. rooftop access, dedicated bridal suite, loading dock…"
              value={form.additionalFeatures}
              onChange={(e) => set("additionalFeatures", e.target.value)}
              className="rounded-xl resize-none"
            />
          </Field>
        </FieldGroup>
      </SectionCard>
    </div>
  );
}

  form,
  setForm,
}: {
  form: VenueFormData;
  setForm: React.Dispatch<React.SetStateAction<VenueFormData>>;
}) {
  const set = <K extends keyof VenueFormData>(k: K, v: VenueFormData[K]) =>
    setForm((f) => ({ ...f, [k]: v }));

  return (
    <div className="space-y-5">
      <SectionCard
        icon={CalendarDays}
        title="Available days"
        subtitle="Which days of the week can guests book?"
      >
        <FieldGroup>
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <Field label="Week opens on" required>
              <Select
                value={form.weekStartDay}
                onValueChange={(v) => set("weekStartDay", v as WeekDay)}
              >
                <SelectTrigger className="rounded-xl">
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  {WEEK_DAYS.map((d) => (
                    <SelectItem key={d.value} value={d.value}>
                      {d.label}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </Field>
            <Field label="Week closes on" required>
              <Select
                value={form.weekEndDay}
                onValueChange={(v) => set("weekEndDay", v as WeekDay)}
              >
                <SelectTrigger className="rounded-xl">
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  {WEEK_DAYS.map((d) => (
                    <SelectItem key={d.value} value={d.value}>
                      {d.label}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </Field>
          </div>

          {/* Visual day range preview */}
          <div className="flex gap-1.5 flex-wrap">
            {(() => {
              const startIdx = WEEK_DAYS.findIndex((d) => d.value === form.weekStartDay);
              const endIdx = WEEK_DAYS.findIndex((d) => d.value === form.weekEndDay);
              return WEEK_DAYS.map((d, i) => {
                const active =
                  startIdx <= endIdx
                    ? i >= startIdx && i <= endIdx
                    : i >= startIdx || i <= endIdx;
                return (
                  <div
                    key={d.value}
                    className={cn(
                      "flex-1 min-w-[36px] py-2 rounded-lg text-center text-xs font-medium transition-colors",
                      active
                        ? "bg-primary text-primary-foreground"
                        : "bg-muted text-muted-foreground",
                    )}
                  >
                    {d.short}
                  </div>
                );
              });
            })()}
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <Field label="Opening time" required>
              <Select
                value={form.venueOpeningTime}
                onValueChange={(v) => set("venueOpeningTime", v || "")}
              >
                <SelectTrigger className="rounded-xl">
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  {TIME_OPTIONS.map((t) => (
                    <SelectItem key={t} value={t}>
                      {t}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </Field>
            <Field label="Closing time" required>
              <Select
                value={form.venueClosingTime}
                onValueChange={(v) => set("venueClosingTime", v || "")}
              >
                <SelectTrigger className="rounded-xl">
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  {TIME_OPTIONS.map((t) => (
                    <SelectItem key={t} value={t}>
                      {t}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </Field>
          </div>
        </FieldGroup>
      </SectionCard>

      <SectionCard
        icon={Clock}
        title="Booking rules"
        subtitle="Control how guests book and cancel"
      >
        <FieldGroup>
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <Field label="Minimum booking duration (hours)" required>
              <Input
                type="number"
                min={1}
                value={form.minDuration}
                onChange={(e) =>
                  set(
                    "minDuration",
                    e.target.value === "" ? "" : Number(e.target.value),
                  )
                }
                className="rounded-xl"
              />
            </Field>
            <Field label="Book in advance (hours)" required>
              <Input
                type="number"
                min={0}
                value={form.bookBefore}
                onChange={(e) =>
                  set(
                    "bookBefore",
                    e.target.value === "" ? "" : Number(e.target.value),
                  )
                }
                className="rounded-xl"
              />
            </Field>
          </div>
        </FieldGroup>
      </SectionCard>

      <SectionCard
        icon={ShieldCheck}
        title="Listing status"
        subtitle="Control visibility on the platform"
      >
        <ToggleRow
          label="List space as active"
          description="Your space will appear in search results immediately after publishing"
          checked={form.isCurrentlyActive}
          onCheckedChange={(v) => set("isCurrentlyActive", v)}
        />
      </SectionCard>
    </div>
  );
}

// ─── Step 4 ───────────────────────────────────────────────────────────────────

function ReviewRow({
  label,
  value,
}: {
  label: string;
  value: React.ReactNode;
}) {
  return (
    <div className="flex justify-between items-start py-2.5 border-b border-border last:border-0 gap-4">
      <span className="text-xs text-muted-foreground shrink-0">{label}</span>
      <span className="text-xs font-medium text-foreground text-right">
        {value || <span className="text-muted-foreground italic">Not set</span>}
      </span>
    </div>
  );
}

function ReviewPublishStep({ form }: { form: VenueFormData }) {
  const venueLabel =
    VENUE_TYPES.find((t) => t.value === form.venueType)?.label ?? "—";
  const startLabel =
    WEEK_DAYS.find((d) => d.value === form.weekStartDay)?.label ?? "—";
  const endLabel =
    WEEK_DAYS.find((d) => d.value === form.weekEndDay)?.label ?? "—";

  return (
    <div className="space-y-5">
      {/* Stat summary row */}
      <div className="grid grid-cols-2 md:grid-cols-3 gap-4">
        {[
          { label: "Capacity", value: form.seatingCapacity || "—", sub: "guests" },
          { label: "Open", value: `${form.venueOpeningTime} – ${form.venueClosingTime}`, sub: "daily hours" },
          { label: "Min. booking", value: form.minDuration ? `${form.minDuration}h` : "—", sub: "minimum" },
        ].map((s) => (
          <div
            key={s.label}
            className="bg-background rounded-2xl border border-border p-4 text-center"
          >
            <p className="text-2xl font-semibold text-foreground">{s.value}</p>
            <p className="text-xs text-muted-foreground mt-0.5">{s.sub}</p>
          </div>
        ))}
      </div>

      <div className="grid grid-cols-1 md:grid-cols-2 gap-5 items-start">
        {/* Left column */}
        <div className="space-y-5">
          <SectionCard icon={Building2} title="Space overview">
            <ReviewRow label="Name" value={form.name} />
            <ReviewRow label="Type" value={venueLabel} />
            <ReviewRow label="City" value={form.city} />
            <ReviewRow label="Address" value={form.address} />
          </SectionCard>

          <SectionCard icon={CalendarDays} title="Availability">
            <ReviewRow label="Open days" value={`${startLabel} – ${endLabel}`} />
            <ReviewRow label="Hours" value={`${form.venueOpeningTime} – ${form.venueClosingTime}`} />
            <ReviewRow label="Book ahead" value={form.bookBefore ? `${form.bookBefore} hours` : "—"} />
            <ReviewRow
              label="Status"
              value={
                <span
                  className={
                    form.isCurrentlyActive
                      ? "text-emerald-600 font-semibold"
                      : "text-muted-foreground"
                  }
                >
                  {form.isCurrentlyActive ? "Active" : "Inactive"}
                </span>
              }
            />
          </SectionCard>
        </div>

        {/* Right column */}
        <div className="space-y-5">
          {/* Photos preview */}
          <SectionCard icon={ImageIcon} title="Photos">
            {form.imagePreviews.length > 0 ? (
              <div className="grid grid-cols-3 gap-2">
                {form.imagePreviews.map((src, i) => (
                  <div
                    key={i}
                    className="relative aspect-square rounded-xl overflow-hidden border border-border"
                  >
                    <img
                      src={src}
                      alt={`Preview ${i + 1}`}
                      className="w-full h-full object-cover"
                    />
                    {i === 0 && (
                      <div className="absolute top-1 left-1">
                        <Badge className="text-[9px] px-1 py-0 bg-primary text-primary-foreground">
                          Cover
                        </Badge>
                      </div>
                    )}
                  </div>
                ))}
              </div>
            ) : (
              <p className="text-xs text-muted-foreground italic">No photos uploaded</p>
            )}
          </SectionCard>

          {/* Amenities */}
          <SectionCard icon={Sparkles} title="Amenities">
            {form.amenities.length > 0 ? (
              <div className="flex flex-wrap gap-2">
                {form.amenities.map((id) => {
                  const a = AMENITIES.find((x) => x.id === id);
                  if (!a) return null;
                  const Icon = a.icon;
                  return (
                    <span
                      key={id}
                      className="flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-primary/10 text-primary text-xs font-medium border border-primary/20"
                    >
                      <Icon className="w-3 h-3" />
                      {a.label}
                    </span>
                  );
                })}
              </div>
            ) : (
              <p className="text-xs text-muted-foreground italic">No amenities selected</p>
            )}
          </SectionCard>

          {/* Publish notice */}
          <div className="flex gap-3 items-start bg-blue-50 dark:bg-blue-950/30 border border-blue-200 dark:border-blue-800 rounded-2xl p-4">
            <Info className="w-4 h-4 text-blue-600 dark:text-blue-400 shrink-0 mt-0.5" />
            <div>
              <p className="text-sm font-semibold text-blue-700 dark:text-blue-300">
                Ready to publish
              </p>
              <p className="text-xs text-blue-600 dark:text-blue-400 mt-1 leading-relaxed">
                Your listing will be reviewed and go live within 24 hours. You
                can edit any details from your dashboard after publishing.
              </p>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}

// ─── Stepper ──────────────────────────────────────────────────────────────────

function VenueStepper({
  step,
  total,
}: {
  step: number;
  total: number;
}) {
  return (
    <div className="flex items-center gap-0">
      {STEPS.map((s, i) => {
        const done = i < step;
        const active = i === step;
        const Icon = s.icon;
        return (
          <React.Fragment key={i}>
            <div className="flex items-center gap-2">
              <div
                className={cn(
                  "flex items-center justify-center w-8 h-8 rounded-full border-2 shrink-0 transition-all text-xs font-semibold",
                  done
                    ? "border-emerald-500 bg-emerald-500 text-white"
                    : active
                    ? "border-primary bg-primary text-primary-foreground"
                    : "border-border bg-background text-muted-foreground",
                )}
              >
                {done ? <Check className="w-3.5 h-3.5" /> : i + 1}
              </div>
              <span
                className={cn(
                  "hidden sm:block text-xs font-medium whitespace-nowrap",
                  active
                    ? "text-foreground"
                    : done
                    ? "text-emerald-600 dark:text-emerald-400"
                    : "text-muted-foreground",
                )}
              >
                {s.label}
              </span>
            </div>
            {i < STEPS.length - 1 && (
              <div
                className={cn(
                  "flex-1 h-px mx-3 transition-colors",
                  i < step ? "bg-emerald-400" : "bg-border",
                )}
              />
            )}
          </React.Fragment>
        );
      })}
      {/* Mobile progress bar */}
      <div className="sm:hidden absolute bottom-0 left-0 right-0 h-0.5 bg-border">
        <div
          className="h-full bg-primary transition-all"
          style={{ width: `${((step + 1) / total) * 100}%` }}
        />
      </div>
    </div>
  );
}

// ─── Main wizard ──────────────────────────────────────────────────────────────

export interface VenueCreationWizardProps {
  /** Control whether the wizard is shown */
  open: boolean;
  onOpenChange: (open: boolean) => void;
  /** Called with collected form data on final "Publish space" click */
  onSubmit?: (data: Omit<VenueFormData, "imagePreviews">) => void | Promise<void>;
}

export function VenueCreationWizard({
  open,
  onOpenChange,
  onSubmit,
}: VenueCreationWizardProps) {
  const [step, setStep] = React.useState(0);
  const [form, setForm] = React.useState<VenueFormData>(INITIAL);
  const [isSubmitting, setIsSubmitting] = React.useState(false);
  const bodyRef = React.useRef<HTMLDivElement>(null);

  const total = STEPS.length;
  const isFirst = step === 0;
  const isLast = step === total - 1;

  const scrollTop = () =>
    bodyRef.current?.scrollTo({ top: 0, behavior: "smooth" });

  const goNext = () => {
    if (!isLast) { setStep((s) => s + 1); scrollTop(); }
  };
  const goBack = () => {
    if (!isFirst) { setStep((s) => s - 1); scrollTop(); }
  };

  const handlePublish = async () => {
    setIsSubmitting(true);
    try {
      // eslint-disable-next-line @typescript-eslint/no-unused-vars
      const { imagePreviews, ...payload } = form;
      await onSubmit?.(payload);
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleClose = () => {
    // Revoke object URLs to avoid memory leaks
    form.imagePreviews.forEach((url) => URL.revokeObjectURL(url));
    setStep(0);
    setForm(INITIAL);
    onOpenChange(false);
  };

  if (!open) return null;

  return (
    // Overlay
    <div
      className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/50 backdrop-blur-sm"
      onClick={(e) => {
        if (e.target === e.currentTarget) handleClose();
      }}
    >
      {/* Dialog panel */}
      <div
        className="
          relative flex flex-col
          w-full max-w-6xl h-[90vh]
          bg-background rounded-2xl
          border border-border shadow-2xl
          overflow-hidden
        "
        role="dialog"
        aria-modal="true"
        aria-label="List a new venue"
      >
        {/* ── Header ──────────────────────────────────────────────── */}
        <div className="relative flex-shrink-0 border-b border-border px-6 md:px-8 pt-6 pb-5 space-y-5">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-3">
              <div className="flex items-center justify-center w-10 h-10 rounded-xl bg-primary/10 shrink-0">
                <Building2 className="w-5 h-5 text-primary" />
              </div>
              <div>
                <h2 className="text-base md:text-lg font-semibold text-foreground leading-none">
                  List a new space
                </h2>
                <p className="text-xs text-muted-foreground mt-0.5">
                  Step {step + 1} of {total} — {STEPS[step].label}
                </p>
              </div>
            </div>
            <button
              onClick={handleClose}
              className="
                flex items-center justify-center w-9 h-9 rounded-lg
                border border-border text-muted-foreground
                hover:bg-accent hover:text-foreground
                transition-colors
              "
              aria-label="Close"
            >
              <X className="w-4 h-4" />
            </button>
          </div>

          <VenueStepper step={step} total={total} />
        </div>

        {/* ── Scrollable body ──────────────────────────────────────── */}
        <div
          ref={bodyRef}
          className="flex-1 overflow-y-auto px-6 md:px-8 py-6 bg-muted/20"
        >
          {step === 0 && <BasicInfoStep form={form} setForm={setForm} />}
          {step === 1 && <MediaFeaturesStep form={form} setForm={setForm} />}
          {step === 2 && <AvailabilityRulesStep form={form} setForm={setForm} />}
          {step === 3 && <ReviewPublishStep form={form} />}
        </div>

        {/* ── Sticky footer ────────────────────────────────────────── */}
        <div className="flex-shrink-0 border-t border-border bg-background px-6 md:px-8 py-4 flex items-center justify-between">
          <p className="text-xs text-muted-foreground hidden sm:block">
            {isLast
              ? "Ready to go — review your details above before publishing"
              : "Fields marked * are required to continue"}
          </p>
          <div className="flex gap-3 ml-auto">
            {!isFirst && (
              <Button
                variant="outline"
                onClick={goBack}
                className="rounded-xl"
                disabled={isSubmitting}
              >
                <ChevronLeft className="w-4 h-4 mr-1" />
                Back
              </Button>
            )}
            {isLast ? (
              <Button
                onClick={handlePublish}
                disabled={isSubmitting}
                className="rounded-xl bg-emerald-600 hover:bg-emerald-700 text-white min-w-[140px]"
              >
                {isSubmitting ? (
                  <>
                    <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                    Publishing…
                  </>
                ) : (
                  <>
                    <Check className="w-4 h-4 mr-2" />
                    Publish space
                  </>
                )}
              </Button>
            ) : (
              <Button
                onClick={goNext}
                className="rounded-xl min-w-[120px]"
              >
                Continue
                <ChevronRight className="w-4 h-4 ml-1" />
              </Button>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
